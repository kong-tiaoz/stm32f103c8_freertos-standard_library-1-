1. 串口高吞吐处理方案 — 三管齐下对抗资源瓶颈

STM32F103C8T6仅20KB RAM，而TAU1201以115200bps持续输出NMEA数据时，每秒约1.5KB。 若不加控制，数秒内就会耗尽内存。以下三重优化确保系统稳定运行。

策略一：按需开关串口

非采样期直接关闭USART3，包括外设时钟和RXNE中断。 仅在接收到信号量后才使能串口，一条GGA采完立即关闭。 中断开销从"持续存在"降为"窗口触发"。
// mycode.c:139-160 GPS_Enable / GPS_Disable static void GPS_Disable(void) { USART_Cmd(GPS_USART, DISABLE); USART_ITConfig(GPS_USART, USART_IT_RXNE, DISABLE); } static void GPS_Enable(void) { // 先清空SR/DR和队列中的残留数据 volatile uint16_t sr = GPS_USART->SR; volatile uint16_t dr = GPS_USART->DR; (void)sr; (void)dr; xQueueReset(gpsQueue); // 清旧数据 USART_ITConfig(GPS_USART, USART_IT_RXNE, ENABLE); USART_Cmd(GPS_USART, ENABLE); }

策略二：仅保留GGA语句

TAU1201默认输出10种NMEA语句（GGA/GLL/GSA/GSV/RMC/VTG/ZDA/GST/TXT等）。 上电后发送10条专有二进制配置指令关闭所有无关输出，数据量降低约90%。
// mycode.c:67-76 UBX协议二进制指令集 const uint8_t close_gll[] = {0xF1,0xD9,0x06,0x01,0x03,0x00,0xF0,0x01,0x00,0xFB,0x11}; const uint8_t close_gsa[] = {0xF1,0xD9,0x06,0x01,0x03,0x00,0xF0,0x02,0x00,0xFC,0x13}; const uint8_t close_rmc[] = {0xF1,0xD9,0x06,0x01,0x03,0x00,0xF0,0x05,0x00,0xFF,0x19}; // ... 共10条指令，逐条发送，每条约11字节 // TAU1201_Config() 连续发送，间隔50ms

策略三：ISR错误处理防死锁

高速串口易发生ORE溢出错误，若ISR只处理RXNE不处理ORE，MCU将陷入无限中断循环。 同时检查所有错误标志，错误字节丢弃但标志位被安全清除。
// mycode.c:199-224 USART3_IRQHandler void USART3_IRQHandler(void) { volatile uint16_t sr = GPS_USART->SR; if(USART_GetITStatus(GPS_USART,USART_IT_RXNE)!=RESET|| (sr&(USART_FLAG_ORE|USART_FLAG_NE| USART_FLAG_FE|USART_FLAG_PE))!=0){ uint8_t ch=USART_ReceiveData(GPS_USART); if((sr&(USART_FLAG_ORE|USART_FLAG_NE| USART_FLAG_FE|USART_FLAG_PE))==0){ xQueueSendFromISR(gpsQueue,&ch,&xHPTW); }else{/* 丢弃错误字节，标志已清除 */} portYIELD_FROM_ISR(xHPTW); } }

4. 异步日志系统 — 不阻塞业务任务的调试输出

核心思路：传统 printf 在115200波特率下发送一行日志可能阻塞数毫秒，对实时系统是致命的。 本系统将日志格式化与物理发送完全解耦。
// bsp_usart.h:43-56 日志系统配置 #define LOG_QUEUE_LENGTH 10 // 队列深度10条 #define LOG_MSG_MAX_LEN 128 // 每条最大128字节 extern bool g_log_output_enabled; // 运行时全局开关 // 核心宏：异步打印，仅格式化+入队，立即返回 #define LOG_PRINT(fmt, ...) Log_Printf(fmt, ##__VA_ARGS__) // bsp_usart.c:44-69 Log_Printf实现 void Log_Printf(const char *format, ...) { if (!g_log_output_enabled) return; // 全局开关 char buffer[LOG_MSG_MAX_LEN]; va_list args; va_start(args, format); vsnprintf(buffer, LOG_MSG_MAX_LEN, format, args); va_end(args); if (xTaskGetSchedulerState() != taskSCHEDULER_NOT_STARTED) { xQueueSendToBack(xLogQueue, buffer, 0); // 非阻塞入队 } else { Usart_SendString(DEBUG_USARTx, buffer); // 调度器未启动时直接发 } }
// bsp_usart.c:26-33 Log_Task 实现 static void Log_Task(void *pvParameters) { char msg[LOG_MSG_MAX_LEN]; while (1) { // 阻塞等待队列有数据 if (xQueueReceive(xLogQueue, msg, portMAX_DELAY) == pdTRUE) { Usart_SendString(DEBUG_USARTx, msg); // 物理发送 } } } // bsp_usart.c:36-41 Log_Init 初始化 void Log_Init(void) { xLogQueue = xQueueCreate(LOG_QUEUE_LENGTH, LOG_MSG_MAX_LEN); if (xLogQueue != NULL) { xTaskCreate(Log_Task, "Log_Task", 128, NULL, 1, &xLogTaskHandle); } }
设计要点：
① LOG_PRINT仅做vsnprintf+入队，不等待发送完成
② 调度器未启动时直接阻塞发送，确保初始化日志不丢失
③ g_log_output_enabled运行时开关，部署后可关闭日志省CPU
④ USART1仅做TX，无RX中断，不占用额外中断资源
